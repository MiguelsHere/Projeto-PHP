CREATE DATABASE workout_db

create table profile(
    id_profile bigint(10) UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    user_name varchar(50) NOT NULL UNIQUE,
    password_hash varchar(255) NOT NULL,
    email varchar(254) NOT NULL UNIQUE,
    date_birth date,
    gender enum('Female','Male','Other','Not Specified') NOT NULL DEFAULT 'Not Specified',
    weight DECIMAL(4,1),
    height DECIMAL(4,1),
    equipment SET('Bench', 'Bands', 'Weight Plates', 'Barbell', 'Dumbell', 'Kettlebell', 'Pull-Up Bar', 'Ab Wheel'),
    profile_description varchar(1000),
    is_certified boolean NOT NULL DEFAULT 0,
    certification_details varchar(1000),
    is_public boolean NOT NULL DEFAULT 0,
    is_active boolean NOT NULL DEFAULT 0,
    created_at datetime DEFAULT CURRENT_TIMESTAMP,
    updated_at datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP

);

create table profile_workout(
    id_profile_workout bigint(10) UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    id_profile bigint(10) UNSIGNED NOT NULL,
    id_workout bigint(10) UNSIGNED NOT NULL,
    is_creator boolean NOT NULL DEFAULT 0,
    rating tinyint(1) UNSIGNED
);

create table workout(
    id_workout bigint(10) UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    workout_name varchar(1000) NOT NULL,
    muscles_worked SET('Calves', 'Quadriceps', 'Hamstrings', 'Glutes') NOT NULL,
    time_needed smallint(3) UNSIGNED,
    created_at datetime DEFAULT CURRENT_TIMESTAMP,
    is_public bool NOT NULL DEFAULT 0
);
